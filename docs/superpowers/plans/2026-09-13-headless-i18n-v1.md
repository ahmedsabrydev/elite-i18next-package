# Headless i18n v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `@faang/i18n` v1 — a headless, typed, i18next-native package with zero-flicker web bootstrapping, shell + lazy namespace loading, and React / Next App Router / Vite adapters — reusable across a React SaaS suite.

**Architecture:** i18next is the engine. The package is a typed kernel (`createI18n`) plus a namespace protocol, a four-tier loader/cache, and thin runtime ports (React, Next, Vite). Locale is URL-first on web. Shell catalogs are inlined at first paint; every other namespace is a lazy chunk. Unused runtimes are separate subpath exports and tree-shake out.

**Tech Stack:** pnpm workspaces + Turborepo · TypeScript 7.0.2 · i18next 26.4.2 · react-i18next 17.0.14 · i18next-cli 1.73.2 · React 19.3.0 · Next 16.3.5 · Vite 8.3.0 · React Router 8.3.1 · Vitest 5.0.0 · Zustand 5.0.15 · RTK 2.12.0

**Spec:** `docs/superpowers/specs/2026-09-13-headless-i18n-monorepo-design.md`

## Global Constraints

- Node.js >= 22 (required by i18next-cli).
- Package manager: pnpm only. Never npm/yarn.
- `"sideEffects": false` in the package manifest.
- `i18next` and `react-i18next` are peer dependencies, never bundled.
- Subpath exports: `/react`, `/next`, `/vite`, `/cli`, `/zustand`, `/redux`, `/native`, `/node`, `/electron`, `/catalogs`.
- Importing `/next` must not statically import Metro, Express, Electron, Zustand, Redux, or `@reduxjs/toolkit`.
- Locale allowlist is exactly `['en', 'ar']` in v1. Any other URL segment resolves to `defaultLocale`. No redirect loops.
- defaultNS: `common`. keySeparator: `.`. nsSeparator: `:`. plural/context separator: `_`.
- JSON is nested objects. Namespace ids and leaf segments must not contain `.` or `:`.
- Namespace id pattern: `/^[a-z][a-zA-Z0-9]*$/`.
- Fallback chain: requested lng -> `en` -> `defaultValue` -> key. `returnEmptyString: false`.
- Budgets (gzip): kernel JS on top of i18next <= 12 KB; shell catalogs per locale <= 12 KB; any single namespace <= 16 KB; manifest <= 1.5 KB.
- Types: TypeScript `strict: true`, `skipLibCheck: true`, `resolveJsonModule: true`.
- `interpolation.escapeValue: true` on non-React surfaces.
- `compatibilityJSON: 'v4'` (only supported value) on i18next and in `CustomTypeOptions`.
- Arabic numbers default to `latn`; `arab` is opt-in per formatter.
- No moment/dayjs/ICU runtime in the kernel.
- Every commit message uses Conventional Commits.
- Every task ends green: `pnpm test` (or the task's stated command) passes before commit.

```ts
import { RTL_LOCALES, SUPPORTED_LOCALES } from './defaults'
import type { DefaultLocale, Dir } from './types'

const SAFE_LNG = /^[a-zA-Z]{2,3}(-[a-zA-Z0-9]{2,8})*$/

export function resolveLocale(
  input: string | null | undefined,
  allowed: readonly string[] = SUPPORTED_LOCALES,
  fallback: string = 'en',
): string {
  if (!input || !SAFE_LNG.test(input)) return fallback
  const base = input.toLowerCase().split('-')[0] ?? ''
  const match = allowed.find((lng) => lng.toLowerCase() === base)
  return match ?? fallback
}

export function dirFor(lng: string): Dir {
  const base = lng.toLowerCase().split('-')[0] ?? ''
  return RTL_LOCALES.has(base) ? 'rtl' : 'ltr'
}

export function assertSupportedLocale<L extends string>(
  lng: string,
  allowed: readonly L[],
): lng is L {
  return allowed.includes(lng as L)
}

export type { DefaultLocale }
```

- [ ] **Step 3: Write `src/cache/lru.ts`**

```ts
export class LruCache<K, V> {
  readonly #capacity: number
  readonly #map = new Map<K, V>()

  constructor(capacity: number) {
    this.#capacity = capacity
  }

  get(key: K): V | undefined {
    if (!this.#map.has(key)) return undefined
    const value = this.#map.get(key) as V
    this.#map.delete(key)
    this.#map.set(key, value)
    return value
  }

  has(key: K): boolean {
    return this.#map.has(key)
  }

  set(key: K, value: V): void {
    if (this.#map.has(key)) this.#map.delete(key)
    this.#map.set(key, value)
    while (this.#map.size > this.#capacity) {
      const oldest = this.#map.keys().next().value
      if (oldest === undefined) break
      this.#map.delete(oldest)
    }
  }

  delete(key: K): boolean {
    return this.#map.delete(key)
  }

  size(): number {
    return this.#map.size
  }

  keys(): K[] {
    return [...this.#map.keys()]
  }
}
```

```ts
import { LOADER } from '../core/defaults'

type InflightOptions = {
  timeoutMs?: number
  retries?: number
  retryDelayMs?: number
}

export class TimeoutError extends Error {
  constructor(key: string, ms: number) {
    super(`i18n load timeout for "${key}" after ${ms}ms`)
    this.name = 'TimeoutError'
  }
}

export class InflightLoader {
  readonly #inflight = new Map<string, Promise<unknown>>()
  readonly #timeoutMs: number
  readonly #retries: number
  readonly #retryDelayMs: number

  constructor(options: InflightOptions = {}) {
    this.#timeoutMs = options.timeoutMs ?? LOADER.timeoutMs
    this.#retries = options.retries ?? LOADER.retries
    this.#retryDelayMs = options.retryDelayMs ?? LOADER.retryDelayMs
  }

  run<T>(key: string, task: () => Promise<T>): Promise<T> {
    const existing = this.#inflight.get(key)
    if (existing) return existing as Promise<T>

    const promise = this.#withRetry(key, task).finally(() => {
      this.#inflight.delete(key)
    })
    this.#inflight.set(key, promise)
    return promise
  }

  size(): number {
    return this.#inflight.size
  }

  async #withRetry<T>(key: string, task: () => Promise<T>): Promise<T> {
    let lastError: unknown
    for (let attempt = 0; attempt <= this.#retries; attempt += 1) {
      try {
        return await this.#withTimeout(key, task)
      } catch (error) {
        lastError = error
        if (attempt < this.#retries) {
          await new Promise((resolve) => setTimeout(resolve, this.#retryDelayMs))
        }
      }
    }
    throw lastError
  }

  #withTimeout<T>(key: string, task: () => Promise<T>): Promise<T> {
    if (this.#timeoutMs <= 0) return task()
    let timer: ReturnType<typeof setTimeout> | undefined
    const timeout = new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(new TimeoutError(key, this.#timeoutMs)), this.#timeoutMs)
    })
    return Promise.race([task(), timeout]).finally(() => {
      if (timer) clearTimeout(timer)
    }) as Promise<T>
  }
}
```

```ts
import type { NamespaceResources } from '../core/types'

const DB_NAME = 'faang-i18n'
const STORE_NAME = 'namespaces'
const DB_VERSION = 1

type CachedRecord = {
  lng: string
  ns: string
  hash?: string
  resources: NamespaceResources
}

export class PersistentStore {
  #dbPromise: Promise<IDBDatabase | undefined> | undefined

  #open(): Promise<IDBDatabase | undefined> {
    if (this.#dbPromise) return this.#dbPromise
    this.#dbPromise = new Promise((resolve) => {
      if (typeof indexedDB === 'undefined') {
        resolve(undefined)
        return
      }
      try {
        const request = indexedDB.open(DB_NAME, DB_VERSION)
        request.onupgradeneeded = () => {
          const db = request.result
          if (!db.objectStoreNames.contains(STORE_NAME)) {
            db.createObjectStore(STORE_NAME, { keyPath: 'id' })
          }
        }
        request.onsuccess = () => resolve(request.result)
        request.onerror = () => resolve(undefined)
      } catch {
        resolve(undefined)
      }
    })
    return this.#dbPromise
  }

  async get(lng: string, ns: string): Promise<NamespaceResources | undefined> {
    const db = await this.#open()
    if (!db) return undefined
    return new Promise((resolve) => {
      try {
        const tx = db.transaction(STORE_NAME, 'readonly')
        const req = tx.objectStore(STORE_NAME).get(`${lng}:${ns}`)
        req.onsuccess = () => {
          const record = req.result as CachedRecord | undefined
          resolve(record?.resources)
        }
        req.onerror = () => resolve(undefined)
      } catch {
        resolve(undefined)
      }
    })
  }

  async set(lng: string, ns: string, resources: NamespaceResources, hash?: string): Promise<void> {
    const db = await this.#open()
    if (!db) return
    await new Promise<void>((resolve) => {
      try {
        const tx = db.transaction(STORE_NAME, 'readwrite')
        tx.objectStore(STORE_NAME).put({ id: `${lng}:${ns}`, lng, ns, hash, resources })
        tx.oncomplete = () => resolve()
        tx.onerror = () => resolve()
      } catch {
        resolve()
      }
    })
  }

  async clear(): Promise<void> {
    const db = await this.#open()
    if (!db) return
    await new Promise<void>((resolve) => {
      try {
        const tx = db.transaction(STORE_NAME, 'readwrite')
        tx.objectStore(STORE_NAME).clear()
        tx.oncomplete = () => resolve()
        tx.onerror = () => resolve()
      } catch {
        resolve()
      }
    })
  }
}
```

```ts
import type { NamespaceResources, ResolvedConfig } from '../core/types'
import { emit } from '../obs/events'
import { InflightLoader, TimeoutError } from './inflight'
import { MemoryStore } from './memory'
import { PersistentStore } from './persistent'

export type NamespaceLoader = (
  lng: string,
  ns: string,
) => Promise<NamespaceResources | undefined>

export type TieredBackendOptions = {
  config: ResolvedConfig
  memory?: MemoryStore
  persistent?: PersistentStore
  inflight?: InflightLoader
  loader?: NamespaceLoader
  fetcher?: typeof fetch
}

export function namespaceChunkLoader(config: ResolvedConfig): NamespaceLoader {
  const { public: publicPath, shared } = config.catalogs
  return async (lng, ns) => {
    const candidates = shared ? [`${shared}/${lng}/${ns}.json`, `${publicPath}/${lng}/${ns}.json`] : [`${publicPath}/${lng}/${ns}.json`]
    for (const specifier of candidates) {
      try {
        const mod = (await import(/* @vite-ignore */ specifier)) as { default?: NamespaceResources } & NamespaceResources
        return (mod.default ?? mod) as NamespaceResources
      } catch {
        // Try the next candidate.
      }
    }
    if (!config.publicPath) return undefined
    try {
      const response = await fetch(`${config.publicPath}/${lng}/${ns}.json`)
      if (!response.ok) return undefined
      return (await response.json()) as NamespaceResources
    } catch {
      return undefined
    }
  }
}

export class TieredBackend {
  readonly #config: ResolvedConfig
  readonly #memory: MemoryStore
  readonly #persistent: PersistentStore
  readonly #inflight: InflightLoader
  readonly #loader: NamespaceLoader

  constructor(options: TieredBackendOptions) {
    this.#config = options.config
    this.#memory = options.memory ?? new MemoryStore()
    this.#persistent = options.persistent ?? new PersistentStore()
    this.#inflight = options.inflight ?? new InflightLoader()
    this.#loader = options.loader ?? namespaceChunkLoader(options.config)
  }

  has(lng: string, ns: string): boolean {
    return this.#memory.has(lng, ns)
  }

  inflightCount(): number {
    return this.#inflight.size()
  }

  async load(lng: string, ns: string): Promise<NamespaceResources | undefined> {
    const cached = this.#memory.get(lng, ns)
    if (cached) {
      emit(this.#config.onEvent, { type: 'ns.cache_hit', lng, ns })
      return cached
    }

    const startedAt = Date.now()
    const key = `${lng}:${ns}`
    try {
      const resources = await this.#inflight.run(key, async () => {
        const fromIdb = await this.#persistent.get(lng, ns)
        if (fromIdb) return fromIdb
        const loaded = await this.#loader(lng, ns)
        if (loaded) await this.#persistent.set(lng, ns, loaded)
        return loaded
      })
      if (!resources) return undefined
      this.#memory.set(lng, ns, resources)
      emit(this.#config.onEvent, {
        type: 'ns.load',
        lng,
        ns,
        ms: Date.now() - startedAt,
        bytes: JSON.stringify(resources).length,
      })
      return resources
    } catch (error) {
      if (error instanceof TimeoutError) {
        emit(this.#config.onEvent, { type: 'ns.timeout', lng, ns, ms: error.message.length })
      }
      return undefined
    }
  }

  prefetch(lng: string, ns: string): void {
    void this.load(lng, ns)
  }
}
```

```ts
import i18next from 'i18next'
import type { NamespaceResources } from './types'
import { DEFAULT_LOCALE } from './defaults'
import { dirFor, resolveLocale } from './locale'
import { resolveConfig } from './resolve-config'
import { TieredBackend } from '../loader/backend'
import type { CreateI18nConfig, DefaultLocale, I18nRuntime, Namespace } from './types'

export function createI18n<L extends string = DefaultLocale>(
  config: CreateI18nConfig<L>,
): I18nRuntime<L> {
  const resolved = resolveConfig(config)
  const backend = new TieredBackend({
    config: resolved,
    loader: resolved.backend
      ? async (lng, ns) => (await resolved.backend?.load(lng, ns)) as NamespaceResources | undefined
      : undefined,
  })

  const instance = i18next.createInstance()
  const initOptions = {
    lng: resolved.defaultLocale,
    fallbackLng: resolved.fallbackLng,
    supportedLngs: [...resolved.locales],
    nonExplicitSupportedLngs: true,
    defaultNS: resolved.defaultNS,
    ns: [resolved.defaultNS],
    keySeparator: '.',
    nsSeparator: ':',
    pluralSeparator: '_',
    contextSeparator: '_',
    compatibilityJSON: 'v4' as const,
    returnEmptyString: false,
    returnNull: false,
    interpolation: { escapeValue: true },
    resources: {},
    initImmediate: false,
  }

  void instance.init(initOptions)

  const loadNamespace = async (lng: L, ns: Namespace): Promise<void> => {
    if (!backend.has(lng, ns)) {
      const resources = await backend.load(lng, ns)
      if (resources) instance.addResourceBundle(lng, ns, resources, true, true)
    }
    if (lng !== resolved.fallbackLng && !backend.has(resolved.fallbackLng, ns)) {
      const fallback = await backend.load(resolved.fallbackLng, ns)
      if (fallback) instance.addResourceBundle(resolved.fallbackLng, ns, fallback, true, true)
    }
  }

  return {
    instance,
    config: resolved,
    loadNamespace,
    prefetch(lng, ns) {
      backend.prefetch(lng, ns)
    },
    has(lng, ns) {
      return backend.has(lng, ns)
    },
    getHtmlAttributes(lng) {
      const safe = resolveLocale(lng, resolved.locales, resolved.defaultLocale)
      return { lang: safe, dir: dirFor(safe) }
    },
    getT(lng, ns = resolved.defaultNS) {
      return instance.getFixedT(lng, ns)
    },
  }
}
```

```js
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { gzipSync } from 'node:zlib'
import { join } from 'node:path'
import { fileURLToPath } from 'node:url'

const root = join(fileURLToPath(new URL('.', import.meta.url)), '..')
const BUDGETS = { namespace: 16 * 1024, shell: 12 * 1024, kernel: 12 * 1024 }
const failures = []

function gzipBytes(path) {
  return gzipSync(readFileSync(path)).byteLength
}

const catalogsDir = join(root, 'catalogs')
for (const lng of readdirSync(catalogsDir)) {
  const dir = join(catalogsDir, lng)
  if (!statSync(dir).isDirectory()) continue
  let shell = 0
  for (const file of readdirSync(dir)) {
    if (!file.endsWith('.json')) continue
    const bytes = gzipBytes(join(dir, file))
    if (file === 'common.json' || file === 'errors.json') shell += bytes
    if (bytes > BUDGETS.namespace) failures.push(`${lng}/${file} ${bytes}B > ${BUDGETS.namespace}B`)
  }
  if (shell > BUDGETS.shell) failures.push(`${lng} shell ${shell}B > ${BUDGETS.shell}B`)
}

const distIndex = join(root, 'dist', 'index.js')
try {
  const bytes = gzipBytes(distIndex)
  if (bytes > BUDGETS.kernel) failures.push(`dist/index.js ${bytes}B > ${BUDGETS.kernel}B`)
} catch {
  failures.push('dist/index.js missing — run build first')
}

if (failures.length) {
  console.error('Budget failures:\n' + failures.map((f) => `  - ${f}`).join('\n'))
  process.exit(1)
}
console.log('Budgets OK')
```

```ts
  changeLocale: (lng: string) => void
} {
  const { snapshot, navigate } = useI18nContext()

  const changeLocale = useMemo(
    () => (lng: string) => {
      if (typeof window === 'undefined') return
      const next = window.location.pathname.replace(/^\/(en|ar)(?=\/|$)/, `/${lng}`)
      const href = next === window.location.pathname ? `/${lng}${window.location.pathname}` : next

      else window.location.assign(href)
    },
    [navigate],
  )

  return { ...snapshot, changeLocale }
}
```

```ts
import { resolveLocale } from '../core/locale'
import type { I18nRuntime } from '../core/types'

export type I18nRequest = {
  pathname: string
  search?: string
  acceptLanguage?: string
  cookie?: string
}

export type I18nRedirect = {
  redirectTo: string
  locale: string
  setCookie: boolean
}

const DEFAULT_COOKIE = 'i18n_lng'

function parseAcceptLanguage(header: string | undefined): string | undefined {
  if (!header) return undefined
  const first = header.split(',')[0]?.trim()
  if (!first) return undefined
  return first.split(';')[0]?.trim()
}

export function i18nMiddleware(
  runtime: I18nRuntime,
  options: { cookieName?: string } = {},
): (request: I18nRequest) => I18nRedirect | undefined {
  const { locales, defaultLocale } = runtime.config
  return (request) => {
    const segments = request.pathname.split('/').filter(Boolean)
    const first = segments[0]
    const hasLocale = first !== undefined && locales.some((lng) => lng.toLowerCase() === first.toLowerCase())
    if (hasLocale) return undefined

    const cookieLng = resolveLocale(request.cookie, locales, defaultLocale)
    const headerLng = resolveLocale(parseAcceptLanguage(request.acceptLanguage), locales, defaultLocale)
    const chosen = request.cookie && cookieLng === request.cookie ? cookieLng : headerLng
    const locale = resolveLocale(chosen, locales, defaultLocale)

    const redirectTo = `/${locale}${request.pathname}${request.search ?? ''}`
    return { redirectTo, locale, setCookie: !request.cookie }
  }
}

export function localeStaticParams(runtime: I18nRuntime): { lng: string }[] {
  return runtime.config.locales.map((lng) => ({ lng }))
}

export { DEFAULT_COOKIE }
```

```ts
import { resolveLocale } from '../core/locale'

export function localePrefix(lng: string, pathname: string): string {
  const normalized = pathname.startsWith('/') ? pathname : `/${pathname}`
  return `/${lng}${normalized}`
}

export function matchLocaleFromPath(
  pathname: string,
  locales: readonly string[],
  defaultLocale: string,
): { locale: string; rest: string } {
  const segments = pathname.split('/')
  const first = segments[1]
  if (!first) return { locale: defaultLocale, rest: pathname }

  const base = first.toLowerCase().split('-')[0] ?? first
  const match = locales.find((lng) => lng.toLowerCase() === base)
  if (!match) return { locale: defaultLocale, rest: pathname }

  const rest = `/${segments.slice(2).join('/')}`.replace(/\/+$/, '')
  return { locale: match, rest }
}

export function resolveLocaleOrNull(
  pathname: string,
  locales: readonly string[],
): string | null {
  const segments = pathname.split('/')
  const first = segments[1]
  if (!first) return null
  const base = first.toLowerCase().split('-')[0] ?? first
  return locales.find((lng) => lng.toLowerCase() === base) ?? null
}

export { resolveLocale }
```

```ts
import { createHash } from 'node:crypto'
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { join } from 'node:path'

export type I18nVitePluginOptions = {
  catalogsDir: string
  locales: readonly string[]
}

export type I18nManifest = {
  version: 1
  locales: Record<string, Record<string, { hash: string; bytes: number }>>
}

export function i18nVitePlugin(options: I18nVitePluginOptions) {
  let manifest: I18nManifest = { version: 1, locales: {} }

  return {
    name: 'faang-i18n',

    buildStart(): void {
      manifest = { version: 1, locales: {} }
      for (const lng of options.locales) {
        const dir = join(options.catalogsDir, lng)
        const entries: Record<string, { hash: string; bytes: number }> = {}
        for (const file of readdirSync(dir)) {
          if (!file.endsWith('.json')) continue
          const full = join(dir, file)
          if (!statSync(full).isFile()) continue
          const source = readFileSync(full)
          entries[file.replace('.json', '')] = {
            hash: createHash('sha256').update(source).digest('hex').slice(0, 12),
            bytes: source.byteLength,
          }
        }
        manifest.locales[lng] = entries
      }
    },

    generateBundle(): { fileName: string; source: string }[] {
      return [
        {
          fileName: 'i18n-manifest.json',
          source: JSON.stringify(manifest),
        },
      ]
    },
  }
}
```

```ts
import { NextResponse, type NextRequest } from 'next/server'
import { i18nMiddleware } from '@faang/i18n/next'
import { runtime } from './i18n'

const middleware = i18nMiddleware(runtime)
const COOKIE = 'i18n_lng'

export function proxy(request: NextRequest) {
  const result = middleware({
    pathname: request.nextUrl.pathname,
    search: request.nextUrl.search,
    acceptLanguage: request.headers.get('accept-language') ?? undefined,
    cookie: request.cookies.get(COOKIE)?.value,
  })
  if (!result) return NextResponse.next()

  const url = request.nextUrl.clone()
  const [pathname, search] = result.redirectTo.split('?')
  url.pathname = pathname ?? '/'
  url.search = search ? `?${search}` : ''
  const response = NextResponse.redirect(url, 307)
  if (result.setCookie) {
    response.cookies.set(COOKIE, result.locale, { path: '/', sameSite: 'lax', httpOnly: false })
  }
  return response
}

export const config = {
  matcher: ['/((?!_next|.*\\..*).*)'],
}
```

```tsx
import { useEffect, useState, type ReactNode } from 'react'
import { useLocation } from 'react-router'
import { dirFor } from '@faang/i18n'
import { matchLocaleFromPath } from '@faang/i18n/vite'
import { runtime } from './i18n'

export function LocaleGuard({ children }: { children: ReactNode }) {
  const location = useLocation()
  const { locale } = matchLocaleFromPath(location.pathname, runtime.config.locales, runtime.config.defaultLocale)
  const [ready, setReady] = useState(() => runtime.has(locale as never, 'common'))

  useEffect(() => {
    document.documentElement.lang = locale
    document.documentElement.dir = dirFor(locale)
  }, [locale])

  useEffect(() => {
    let cancelled = false
    if (!runtime.has(locale as never, 'common')) {
      void runtime.loadNamespace(locale as never, 'common').then(() => {
        if (!cancelled) setReady(true)
      })
    }
    return () => {
      cancelled = true
    }
  }, [locale])

  if (!ready) return <div data-testid="locale-boot" aria-busy="true" />

  return <>{children}</>
}
```
